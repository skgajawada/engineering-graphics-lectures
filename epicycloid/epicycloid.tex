\begin{frame}{Problem Statement}
\vfill
\begin{block}{Question}
Draw an \textbf{epicycloid} generated by a circle of diameter $D=50\text{ mm}$ rolling externally without slipping on a fixed circle of diameter $D_f=100\text{ mm}$ for \textbf{one complete revolution of the generating-circle centre}.
Construct a \textbf{tangent} and a \textbf{normal} to the epicycloid at a selected point $P$.
\end{block}
\begin{alertblock}{Given}
$R=50\text{ mm},\qquad r=25\text{ mm}$
\end{alertblock}
\vfill
