\begin{frame}{Problem Statement}

\begin{block}{Given}

Radius of fixed circle:

\[
R=50\text{ mm}
\]

Radius of generating circle:

\[
r=25\text{ mm}
\]

The generating circle rolls externally without slipping around
the fixed circle.

\end{block}

\begin{block}{Required}

Construct the epicycloid for one complete revolution of the
generating-circle centre and draw the tangent and normal at a
selected point $P$.

\end{block}

\end{frame}

\begin{frame}{Step 5: Determine the Rolling Rotation}

\begin{block}{No-Slip Condition}

When the generating circle rolls externally,

\[
(R+r)\theta=r\phi
\]

Therefore,

\[
\boxed{
\phi=\frac{R+r}{r}\theta
}
\]

For

\[
R=50\text{ mm},\qquad r=25\text{ mm},
\]

\[
\boxed{\phi=3\theta}
\]

Thus, when the centre completes one revolution,

\[
\theta=360^\circ
\]

the generating circle rotates through

\[
\boxed{\phi=1080^\circ}
\]

or three complete rotations.

\end{block}

\end{frame}

\begin{frame}{Step 7: Locate the Epicycloid Points}

\centering

\begin{tikzpicture}[scale=0.62]

    \coordinate (O) at (0,0);

    \draw[fixedcircle] (O) circle (3);
    \draw[locus] (O) circle (4.5);

    \foreach \a [count=\i] in {0,30,...,330}

        \coordinate (C)
            at ({4.5*cos(\a)},{4.5*sin(\a)});

        % Epicycloid point
        \pgfmathsetmacro{\px}
            {4.5*cos(\a)-1.5*cos(3*\a)}

        % Radius construction
        \draw[construction]

        \fill (P) circle (2pt);

\end{tikzpicture}

\begin{frame}{Step 8: Complete the Epicycloid}

\centering

\begin{tikzpicture}[scale=0.62]

    \coordinate (O) at (0,0);

    \draw[fixedcircle] (O) circle (3);

    \draw[locus]
        (O) circle (4.5);

    % Epicycloid
    \draw[
        curve,
        samples=300,
        domain=0:360,
        variable=\t
    ]
    plot
    (
        {4.5*cos(\t)-1.5*cos(3*\t)},
        {4.5*sin(\t)-1.5*sin(3*\t)}
    );

    % Construction points
    \foreach \a in {0,30,...,330}
    {
        \pgfmathsetmacro{\px}
            {4.5*cos(\a)-1.5*cos(3*\a)}

        \pgfmathsetmacro{\py}
            {4.5*sin(\a)-1.5*sin(3*\a)}

        \fill
            ({\px},{\py})
            circle (1.5pt);
    }

    \fill (O) circle (2pt)
        node[below left] {$O$};

\end{tikzpicture}

\[
\boxed{
x=(R+r)\cos\theta-r\cos3\theta
}
\]

\[
\boxed{
y=(R+r)\sin\theta-r\sin3\theta
}
\]

\end{frame}

\begin{frame}{Step 10: Draw the Normal}

\centering

\begin{tikzpicture}[scale=0.68]

    \coordinate (O) at (0,0);

    \draw[fixedcircle] (O) circle (3);

    \draw[locus]
        (O) circle (4.5);

    \draw[
        curve,
        samples=300,
        domain=0:360,
        variable=\t
    ]
    plot
    (
        {4.5*cos(\t)-1.5*cos(3*\t)},
        {4.5*sin(\t)-1.5*sin(3*\t)}
    );

    \pgfmathsetmacro{\th}{30}

    \coordinate (M)
        at ({3*cos(\th)},{3*sin(\th)});

    \pgfmathsetmacro{\px}
        {4.5*cos(\th)-1.5*cos(3*\th)}

    \pgfmathsetmacro{\py}
        {4.5*sin(\th)-1.5*sin(3*\th)}

    \coordinate (P) at (\px,\py);

    % Direction of normal = P-M
    \pgfmathsetmacro{\nx}{\px-3*cos(\th)}
    \pgfmathsetmacro{\ny}{\py-3*sin(\th)}

    \pgfmathsetmacro{\nl}
        {sqrt(\nx*\nx+\ny*\ny)}

    \pgfmathsetmacro{\ux}{\nx/\nl}
    \pgfmathsetmacro{\uy}{\ny/\nl}

    % Normal
    \draw[
        normal
    ]
    ({\px-3*\ux},{\py-3*\uy})
        node[below left] {$N$}
    --
    ({\px+3*\ux},{\py+3*\uy})
        node[above right] {$N'$};

    % PM
    \draw[construction]
        (M)--(P);

    \fill (P) circle (3pt)
        node[above right] {$P$};

    \fill (M) circle (3pt)
        node[below left] {$M$};

\end{tikzpicture}

\begin{block}{Normal}
The normal at $P$ passes through the instantaneous contact
point $M$.

\[
\boxed{MP\text{ is the normal}}
\]

\end{block}

\end{frame}

The tangent is perpendicular to the normal at $P$.

\[
\boxed{TT'\perp NN'}
\]

\end{block}
