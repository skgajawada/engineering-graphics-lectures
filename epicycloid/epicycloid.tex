\begin{frame}{Problem Statement}
  \vfill
  \begin{block}{Question}
    \vspace{0.8em}
    Draw an \textbf{epicycloid} generated by a rolling circle of diameter $d = 40\text{ mm}$ rolling on the outside of a fixed base circle of diameter $D = 80\text{ mm}$ for one complete revolution.
    \vspace{0.8em}
    Construct a \textbf{tangent} and a \textbf{normal} at any point $P$ on the curve.
  \end{block}
  \vfill
\end{frame}

\renewcommand{\stepnum}{1}
\begin{frame}{Step 1: Draw Fixed Base Circle \& Rolling Circle}
\centering
\begin{tikzpicture}[scale=0.45]
  \useasboundingbox (-6.5,-6.5) rectangle (18.85,10.0);
  \drawLegendCard{ \textbf{Base Circle} & Radius $R = 40$ \\ \textbf{Rolling Circle} & Radius $r = 20$ }
  \coordinate (O) at (0,0);
  \draw[thick, blue!80!black] (O) circle (4.0);
  \fill[red!80!black] (O) circle (3.5pt) node[below left] {$O$};
  \coordinate (C0) at (0,6.0);
  \draw[thick, blue!80!black] (C0) circle (2.0);
  \fill[red!80!black] (C0) circle (3.5pt) node[above] {$C_0$};
  \draw[axis line] (0,0) -- (0,8.0);
\end{tikzpicture}
\end{frame}

\renewcommand{\stepnum}{2}
\begin{frame}{Step 2: Complete Epicycloid Curve with Tangent \& Normal}
\centering
\begin{tikzpicture}[scale=0.45]
  \useasboundingbox (-6.5,-6.5) rectangle (18.85,10.0);
  \drawLegendCard{ \textbf{Epicycloid} & Generated Curve \\ \textbf{$N\text{-}N'$} & Normal Line \\ \textbf{$T\text{-}T'$} & Tangent Line }
  \coordinate (O) at (0,0);
  \draw[thick, blue!80!black] (O) circle (4.0);
  \fill[red!80!black] (O) circle (3.5pt) node[below left] {$O$};
  
  \draw[main curve, samples=100, domain=0:6.28318] plot (
    {(4+2)*sin(\x r) - 2*sin((4+2)*\x / 2 r)},
    {(4+2)*cos(\x r) - 2*cos((4+2)*\x / 2 r)}
  );
  
  \coordinate (P) at (4.5, 6.2);
  \fill[black] (P) circle (3.5pt) node[above right] {$P$};
  \coordinate (Cp) at (2.5, 5.5);
  \draw[construction line] (P) -- (Cp);
  \fill[red!80!black] (Cp) circle (3.5pt) node[above] {$C_p$};
  \draw[normal line] (O) -- (Cp) -- (P);
  \draw[tangent line] (1.2, 8.5) node[above] {$T$} -- (7.5, 3.2) node[below] {$T'$};
\end{tikzpicture}
\end{frame}
