\begin{frame}{Problem Statement}

  \vfill

  \begin{block}{Question}
    \vspace{0.8em}

    Draw an \textbf{epicycloid} generated by a circle of diameter

    \[
    D=50\text{ mm}
    \]

    rolling externally without slipping on a fixed circle of diameter

    \[
    D_f=100\text{ mm}
    \]

    for \textbf{one complete revolution of the generating-circle centre}.

    \vspace{0.8em}

    Construct a \textbf{tangent} and a \textbf{normal} to the
    epicycloid at a selected point \(P\).

  \end{block}

  \vfill

\end{frame}

\renewcommand{\stepnum}{6}

\begin{frame}{Step 6: Determine the Rolling Relation}

\begin{block}{No-Slip Condition}

For external rolling,

\[
(R+r)\theta=r\phi
\]

Hence,

\[
\boxed{
\phi=\frac{R+r}{r}\theta
}
\]

For

\[
R=50\text{ mm},
\qquad
r=25\text{ mm},
\]

\[
\phi=
\frac{75}{25}\theta
=
3\theta.
\]

Therefore,

\[
\boxed{\phi=3\theta}
\]

\end{block}

\begin{block}{For 12 divisions}

The centre moves by

\[
30^\circ
\]

for every step.

Therefore the generating circle rotates through

\[
3(30^\circ)=90^\circ.
\]

\[
\boxed{\text{One centre division } \Rightarrow
\text{ three generating-circle divisions}}
\]

\end{block}

\end{frame}

  \phi=3\theta.

  \item Locate
  \[
  P_0,P_1,\ldots,P_{12}.

  \item For the selected point \(P\), locate the instantaneous contact
  point \(M\).

  \boxed{MP=NN'}
  \]

  \item Draw the tangent perpendicular to the normal:

  \]

\end{enumerate}
