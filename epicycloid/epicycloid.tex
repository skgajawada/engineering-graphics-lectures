\begin{frame}{Problem Statement}
\vfill
\begin{block}{Question}
Draw an \textbf{epicycloid} generated by a circle of diameter $D=50\text{ mm}$ rolling externally without slipping on a fixed circle of diameter $D_f=100\text{ mm}$ for \textbf{one complete revolution of the generating-circle centre}.
Construct a \textbf{tangent} and a \textbf{normal} to the epicycloid at a selected point $P$.
\end{block}
\begin{alertblock}{Given}
$R=50\text{ mm},\qquad r=25\text{ mm}$
\end{alertblock}
\vfill
\end{frame}

% ============================================================
% STEP 5: NO-SLIP CONDITION & ROTATION FORMULA
% ============================================================
\renewcommand{\stepnum}{5}
\begin{frame}{Step 5: Establish No-Slip Condition \& Rotation Ratio}
\centering
\begin{block}{Rolling relation}
\[(R+r)\theta=r\phi \implies \phi=\frac{R+r}{r}\theta = \frac{50+25}{25}\theta = 3\theta\]
\end{block}
\begin{itemize}
  \item Every $30^\circ$ centre movement $\rightarrow$ Rolling circle rotates by $3\times30^\circ = 90^\circ$.
  \item 3 rolling-circle divisions per centre step.
\end{itemize}
\end{frame}
