\begin{frame}{Problem Statement}
\begin{block}{Given}
A generating circle of diameter $30$ mm rolls externally without slipping on a directing circle of diameter $90$ mm.
\end{block}
\begin{itemize}
\item Construct one complete epicycloid arch. $\angle AOB=120^\circ$ and the arch is divided into 12 equal angular parts. Retain every construction element in all later views.
\end{itemize}
\end{frame}

\begin{frame}{Calculated Data}
\begin{align*}
R&=90/2=45\text{ mm},&r&=30/2=15\text{ mm},\\
OC_0&=R+r=60\text{ mm},&AC_0&=15\text{ mm},\\
\angle AOB&=120^\circ,&\Delta\theta&=120^\circ/12=10^\circ.
\end{align*}
\end{frame}
