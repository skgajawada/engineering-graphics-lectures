% ------------------------------------------------------------
% Generic construction arc
% ------------------------------------------------------------

\newcommand{\ConArcCustom}[1]{%

  \pgfmathsetmacro{\gx}{
    60*cos(135)+15*cos(135-30*(#1))
  }

  \pgfmathsetmacro{\gy}{
    60*sin(135)+15*sin(135-30*(#1))
  }

  \pgfmathsetmacro{\rr}{
    sqrt(\gx*\gx+\gy*\gy)
  }

  \pgfmathsetmacro{\aa}{
    atan2(\gy,\gx)
  }

  \draw[arc line]
    (\aa:\rr)
    arc[
      start angle=\aa,
      end angle=63,
      radius=\rr
    ];
}

% ============================================================

\begin{frame}{Problem Statement}

\begin{block}{Given}

A generating circle of diameter $30$ mm rolls internally
without slipping inside a directing circle of diameter
$150$ mm.

\end{block}

\begin{itemize}

  \item Construct one complete hypocycloid arch.

  \item Divide the arch into 12 equal angular parts.

  \item Retain every construction element in all later views.

\end{itemize}

\end{frame}

% ============================================================
% CALCULATED DATA
% ============================================================

\begin{frame}{Calculated Data}

\begin{align*}
R
&=
\frac{150}{2}
=
75\text{ mm},
&
r
&=
\frac{30}{2}
=
15\text{ mm},
\\[0.3em]
OC_0
&=
R-r
=
75-15
=
60\text{ mm},
&
AC_0
&=
15\text{ mm},
\\[0.3em]
\angle AOB
&=
\frac{360^\circ r}{R}
=
\frac{360^\circ(15)}{75}
=
72^\circ,
\\[0.3em]
\Delta\theta
&=
\frac{72^\circ}{12}
=
6^\circ,
&
\text{generating-circle division}
&=
\frac{360^\circ}{12}
=
30^\circ.
\end{align*}

\end{frame}
