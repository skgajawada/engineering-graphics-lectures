%
%   1 & 11
%   2 & 10
%   3 & 9
%   4 & 8
%   5 & 7
%   6 alone
%
% Only ONE physical arc is drawn for each pair.
% ============================================================

\newcommand{\ConArcPair}[2]{%

    % Coordinates of generating-circle division #1
    \pgfmathsetmacro{\gx}{
        60*cos(135)+25*cos(135-30*(#1))
    }

    \pgfmathsetmacro{\gy}{
        60*sin(135)+25*sin(135-30*(#1))
    }

    % Radius from O to generating-circle point
    \pgfmathsetmacro{\rr}{
        sqrt(\gx*\gx+\gy*\gy)
    }

    % Polar angle of that point
    \pgfmathsetmacro{\aa}{
        atan2(\gy,\gx)
    }

    % Single physical concentric arc
    \draw[arc line]
        (\aa:\rr)
        arc (\aa:\Bang:\rr);
}

\begin{frame}{Problem Statement}

\begin{block}{Problem}

Draw a hypocycloid of a circle of diameter 50 mm, which rolls
inside a circular arc of radius 85 mm for one revolution. Also,
draw a tangent and a normal to the epicycloid at a point 50 mm
from the centre of the directing circle.

\end{block}

\end{frame}

% ============================================================
% CALCULATED DATA
% ============================================================

\begin{frame}{Calculated Data}

\begin{align*}

R
&=85\text{ mm},
&
r
&=\frac{50}{2}=25\text{ mm},
\\[0.5em]

OC_0
&=R-r
=85-25
=60\text{ mm},
&
AC_0
&=25\text{ mm},
\\[0.5em]

\angle AOB
&=
360^\circ\frac{r}{R}
=
360^\circ\frac{25}{85}
=
105.882353^\circ,
\\[0.5em]

\Delta\theta
&=
\frac{105.882353^\circ}{12}
=
8.823529^\circ,
&
\text{Generating-circle division}
&=30^\circ.
\\[0.5em]

\frac{R-r}{r}
&=
\frac{60}{25}
=
2.4.
\end{align*}

\end{frame}
