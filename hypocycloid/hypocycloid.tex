\newcommand{\ConArcZero}{%

        arc line
    ]
    (135:75)

\newcommand{\ConArcCustom}[1]{%

    % Coordinates of G_i
    \pgfmathsetmacro{\gx}{
        60*cos(135)
        +
        15*cos(135-30*(#1))
    }

    \pgfmathsetmacro{\gy}{
        60*sin(135)
        +
        15*sin(135-30*(#1))
    }

    % Radius OG_i
    \pgfmathsetmacro{\rr}{
        sqrt(
            \gx*\gx+\gy*\gy
        )
    }

    % Angular position of G_i from O
    \pgfmathsetmacro{\aa}{
        atan2(\gy,\gx)
    }

    % Concentric arc
    \draw[
        arc line
    ]
    (\aa:\rr)
    arc[
        start angle=\aa,
        end angle=63,
        radius=\rr
    ];
}

\newcommand{\ConFull}{%

    \ConArcZero

    \foreach\k in {1,...,12}{
        \ConArcCustom{\k}
    }
}

% ============================================================
% HYPOCYCLOID POINTS
%
% Centre angle:
%
%       alpha = 135 - 6 theta
%
% Radius-vector angle:
%
%       beta = 135 + 24 theta
%
% because
%
%       (R-r)/r = 60/15 = 4
%
% and the centre moves clockwise.
%
% With i = 0,...,12:
%
%       alpha_i = 135 - 6i
%       beta_i  = 135 + 24i
%
% ============================================================

\newcommand{\AllPCoords}{%

    \foreach\k in {0,...,12}{

        \coordinate(P\k)
        at (
            {
                60*cos(135-6*(\k))

                15*cos(135+24*(\k))
            },
            {
                60*sin(135-6*(\k))
                +
                15*sin(135+24*(\k))
            }
        );
    }
}

\begin{frame}{Problem Statement}

\begin{block}{Given}

A generating circle of diameter $30$ mm rolls internally
without slipping inside a directing circle of diameter
$150$ mm.

\end{block}

\begin{itemize}

    \item Construct one complete hypocycloid arch.

    \item Divide the arch into 12 equal angular parts.

    \item Retain every construction element in all later views.

\end{itemize}

\end{frame}

% ============================================================
% CALCULATED DATA
% ============================================================

\begin{frame}{Calculated Data}

\begin{align*}

R
&=
\frac{150}{2}
=
75\text{ mm},
&
r
&=
\frac{30}{2}
=
15\text{ mm},
\\[0.4em]

OC_0
&=
R-r
=
75-15
=
60\text{ mm},
&
AC_0
&=
15\text{ mm},
\\[0.4em]

\angle AOB
&=
\frac{360^\circ r}{R}
=
\frac{360^\circ(15)}{75}
=
72^\circ,
\\[0.4em]

\Delta\theta
&=
\frac{72^\circ}{12}
=
6^\circ,
&
\text{Generating-circle division}
&=
\frac{360^\circ}{12}
=
30^\circ.
\end{align*}

\end{frame}
