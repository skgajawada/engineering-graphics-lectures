\begin{frame}{Problem Statement}
\begin{block}{Given}
A generating circle of diameter $50$ mm rolls internally without slipping inside a directing circular arc of radius $85$ mm for one revolution.
\end{block}
\begin{itemize}
    \item Construct the hypocycloid arch.
    \item Draw a tangent and normal at a point $50$ mm from the centre of the directing circle.
    \item Retain every construction element in all later views.
\end{itemize}
\end{frame}

\begin{frame}{Calculated Data}
\begin{align*}
R &= 85\text{ mm},
&
r &= \frac{50}{2}=25\text{ mm},
\\[0.4em]
OC_0 &= R-r=85-25=60\text{ mm},
&
AC_0 &= 25\text{ mm},
\\[0.4em]
\angle AOB &= 360^\circ \times \frac{r}{R} = 360^\circ \times \frac{25}{85} \approx 105.88^\circ,
\\[0.4em]
\Delta\theta &= \frac{105.88^\circ}{12} \approx 8.8235^\circ,
&
\text{Generating-circle division} &= \frac{360^\circ}{12} = 30^\circ.
\end{align*}
\end{frame}
