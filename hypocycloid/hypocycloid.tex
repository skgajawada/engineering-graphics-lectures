\begin{frame}{Problem Statement}
\begin{block}{Problem}
\vspace{0.55em}
Construct one complete hypocycloid arch generated by a circle of diameter $30$ mm rolling internally, without slipping, inside a fixed/directing circle of diameter $150$ mm. The starting point of contact is $A$, the angle $AOB=120^\circ$, and the arch is to be divided into 12 equal angular parts. Construct the successive positions of the generating circle and tracing point, and construct the tangent and normal at a selected point $P$ on the hypocycloid.
\vspace{0.55em}
\end{block}
\end{frame}

\begin{frame}{Given Data and Calculations}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\textbf{Given data}
\begin{align*}
\text{Diameter of directing circle}&=150\text{ mm},\\
R&=75\text{ mm},\\
\text{Diameter of generating circle}&=30\text{ mm},\\
r&=15\text{ mm},\\
\angle AOB&=120^\circ,\\
\text{Let number of divisions}&=12.
\end{align*}
\vspace{-0.35em}
\textbf{Calculations}
\begin{align*}
OC_0&=R-r=75-15=60\text{ mm},\\
AC_0&=r=15\text{ mm},\\
\Delta\theta&=\frac{120^\circ}{12}=10^\circ.
\end{align*}

\column{0.49\textwidth}
\textbf{Generating-circle division}
\begin{align*}
\text{Division angle}&=\frac{360^\circ}{12}=30^\circ.
\end{align*}
\vspace{0.3em}
\textbf{Rolling relation}
\begin{align*}
\text{Each }10^\circ\text{ centre step}
&\leftrightarrow 30^\circ\text{ generating-circle step}.
\end{align*}
\vspace{0.3em}
\textbf{Centre locus}
\begin{align*}
OC&=R-r=60\text{ mm}.
\end{align*}
\end{columns}
\end{frame}
