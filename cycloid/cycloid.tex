% --- SLIDE 2 ---
\begin{frame}[fragile]{Definition and Applications}{Radius of generating circle: $R = 20\text{ mm}$}
\begin{itemize}
    \item Locus of a point on the circumference of a circle rolling along a straight line.
    \item Widely used in gear tooth profiles (cycloidal gearing).
    \item Isochrone curve properties in physics and mechanics.
\end{itemize}
\end{frame}

% --- SLIDE 3 ---
\begin{frame}[fragile]{Required Parameters}{Radius of generating circle: $R = 20\text{ mm}$}
\begin{itemize}
    \item Radius of generating circle ($R$) = $20\text{ mm}$.
    \item Circumference of the rolling circle $\pi \times D = 2 \pi R \approx 125.66\text{ mm}$.
    \item Division of circle into 12 equal angular parts ($30^\circ$ each).
    \item Horizontal linear displacement divided into 12 equal intervals.
\end{itemize}
\end{frame}
